    function filterCategory(cat, btnElement) {
      currentCategory = cat;
      document.querySelectorAll('.cat-btn').forEach(btn => {
        btn.className = "cat-btn bg-white text-slate-700 border border-emerald-900/10 px-4 py-2 rounded-full text-xs font-bold transition hover:bg-emerald-50";
      });
      btnElement.className = "cat-btn bg-[#1b4332] text-amber-200 border border-[#c5a880]/40 px-5 py-2 rounded-full text-xs font-bold transition shadow-sm";
      renderProducts();
    }

    function toggleCartDrawer() {
      document.getElementById('cart-drawer').classList.toggle('hidden');
      renderCart();
    }

    function addToCart(id) {
      const prod = products.find(p => p.id === id);
      if (!prod) return;

      const existing = cart.find(item => item.id === id);
      if (existing) {
        existing.qty += 1;
      } else {
        cart.push({ id: prod.id, title: prod.title, price: prod.price, qty: 1 });
      }

      saveCart();
      updateCartBadge();
    }

    function updateCartQty(id, delta) {
      const item = cart.find(i => i.id === id);
      if (!item) return;

      item.qty += delta;
      if (item.qty <= 0) {
        cart = cart.filter(i => i.id !== id);
      }

      saveCart();
      renderCart();
      updateCartBadge();
    }

    function saveCart() {
      localStorage.setItem('lavera_cart_data', JSON.stringify(cart));
    }

    function updateCartBadge() {
      const count = cart.reduce((sum, item) => sum + item.qty, 0);
      document.getElementById('cart-counter').innerText = count;
    }

    function renderCart() {
      const container = document.getElementById('cart-items-list');
      const totalEl = document.getElementById('cart-total-price');

      if (cart.length === 0) {
        container.innerHTML = `<div class="text-center py-12 text-slate-400 font-bold text-sm">سلة المشتريات فارغة حالياً</div>`;
        totalEl.innerText = "0 ج.م";
        return;
      }

      let total = 0;
      container.innerHTML = cart.map(item => {
        const itemTotal = item.price * item.qty;
        total += itemTotal;
        return `
          <div class="flex items-center justify-between p-3 bg-slate-50 rounded-xl border border-slate-200/80">
            <div class="flex-grow pr-2">
              <h4 class="font-bold text-slate-800 text-sm leading-tight mb-1">${item.title}</h4>
              <span class="text-xs text-slate-500">${item.price} ج.م &times; ${item.qty} = <strong>${itemTotal} ج.م</strong></span>
            </div>
            <div class="flex items-center gap-2">
              <button onclick="updateCartQty('${item.id}', -1)" class="w-7 h-7 bg-white border border-slate-200 rounded-lg flex items-center justify-center text-slate-600 hover:bg-slate-100 font-bold">-</button>
              <span class="text-sm font-bold w-4 text-center">${item.qty}</span>
              <button onclick="updateCartQty('${item.id}', 1)" class="w-7 h-7 bg-white border border-slate-200 rounded-lg flex items-center justify-center text-slate-600 hover:bg-slate-100 font-bold">+</button>
            </div>
          </div>
        `;
      }).join('');

      totalEl.innerText = `${total} ج.م`;
    }

    function completeOrderWhatsApp() {
      if (cart.length === 0) {
        alert("السلة فارغة!");
        return;
      }

      const name = document.getElementById('order-name').value.trim();
      const phone = document.getElementById('order-phone').value.trim();
      const address = document.getElementById('order-address').value.trim();

      if (!name || !phone || !address) {
        alert("يرجى ملء جميع بيانات التوصيل (الاسم، الهاتف، والعنوان بالتفصيل).");
        return;
      }

      let total = 0;
      let itemsSummary = cart.map((i, idx) => {
        const sub = i.price * i.qty;
        total += sub;
        return `${idx + 1}. ${i.title} (${i.qty} قطعة) = ${sub} ج.م`;
      }).join('\n');

      let message = `🛒 *طلب شراء جديد من متجر LAVÉRA*\n\n` +
                    `👤 *الاسم:* ${name}\n` +
                    `📞 *الهاتف:* ${phone}\n` +
                    `📍 *العنوان:* ${address}\n\n` +
                    `📦 *تفاصيل الطلب:*\n${itemsSummary}\n\n` +
                    `💰 *المبلغ الإجمالي:* ${total} ج.م\n` +
                    `يرجى تأكيد الطلب وتحديد موعد الاستلام.`;

      const encodedMsg = encodeURIComponent(message);
      window.open(`https://wa.me/${STORE_WHATSAPP}?text=${encodedMsg}`, '_blank');
    }

    function authenticateAdmin() {
      const pass = document.getElementById('admin-pass-input').value;
      const savedPass = localStorage.getItem('lavera_admin_pass') || '1234';

      if (pass === savedPass) {
        isAdmin = true;
        document.getElementById('admin-login-view').classList.add('hidden');
        document.getElementById('admin-panel-view').classList.remove('hidden');
        document.getElementById('admin-pass-input').value = '';
        renderProducts();
      } else {
        alert("كلمة المرور غير صحيحة!");
      }
    }

    }

    function saveOrUpdateProduct() {
      const editId = document.getElementById('edit-prod-id').value;
      const title = document.getElementById('adm-title').value.trim();
      const sub = document.getElementById('adm-sub').value.trim();
      const cat = document.getElementById('adm-cat').value;
      const price = parseFloat(document.getElementById('adm-price').value) || 0;
      const size = document.getElementById('adm-size').value.trim();
      const badge = document.getElementById('adm-badge').value.trim();
      const itemsRaw = document.getElementById('adm-items').value.trim();
      const items = itemsRaw ? itemsRaw.split('\n').filter(t => t.trim().length > 0) : [];

      if (!title || price <= 0) {
        alert("يرجى إدخال اسم المنتج وسعر صحيح.");
        return;
      }

      if (editId) {
        const p = products.find(prod => prod.id === editId);
        if (p) {
          p.title = title;
          p.sub = sub;
          p.cat = cat;
          p.price = price;
          p.size = size;
          p.badge = badge;
          p.items = items;
          if (uploadedImageData) p.image = uploadedImageData;
        }
      } else {
        const newProduct = {
          id: Date.now().toString(),
          title,
          sub,
          cat,
          price,
          size,
          badge,
          items,
          image: uploadedImageData || "https://images.unsplash.com/photo-1584308666744-24d5c474f2ae?auto=format&fit=crop&w=600&q=80"
        };
        products.push(newProduct);
      }

      localStorage.setItem('lavera_products_data', JSON.stringify(products));
      renderProducts();
      cancelEditMode();
      alert("تم حفظ بيانات المنتج والصورة بنجاح!");
    }

    function editProductPrompt(id) {
      const p = products.find(prod => prod.id === id);
      if (!p) return;

      openAdminModal();
      document.getElementById('edit-prod-id').value = p.id;
      document.getElementById('adm-title').value = p.title || '';
      document.getElementById('adm-sub').value = p.sub || '';
      document.getElementById('adm-cat').value = p.cat || 'CLOTHS';
      document.getElementById('adm-price').value = p.price || '';
      document.getElementById('adm-size').value = p.size || '';
      document.getElementById('adm-badge').value = p.badge || '';
      document.getElementById('adm-items').value = (p.items || []).join('\n');
      uploadedImageData = "";
      document.getElementById('img-preview-info').classList.add('hidden');

      document.getElementById('admin-form-title').innerText = "✏️ تعديل بيانات المنتج:";
      document.getElementById('admin-cancel-btn').classList.remove('hidden');
    }

    function cancelEditMode() {
      document.getElementById('edit-prod-id').value = '';
      document.getElementById('adm-title').value = '';
      document.getElementById('adm-sub').value = '';
      document.getElementById('adm-price').value = '';
      document.getElementById('adm-size').value = '';
      document.getElementById('adm-badge').value = '';
      document.getElementById('adm-items').value = '';
      const fileInput = document.getElementById('adm-img-file');
      if (fileInput) fileInput.value = '';
      uploadedImageData = "";

      document.getElementById('admin-cancel-btn').classList.add('hidden');
    }

    function deleteProduct(id) {
      if (confirm("هل أنت متأكد من حذف هذا المنتج نهائياً من الكتالوج؟")) {
        products = products.filter(p => p.id !== id);
        localStorage.setItem('lavera_products_data', JSON.stringify(products));
        renderProducts();
      }
    }

    function changeAdminPassword() {
      const newPass = document.getElementById('new-admin-pass').value.trim();
      if (newPass.length < 4) {
        alert("كلمة المرور يجب ألا تقل عن 4 أحرف/أرقام.");
        return;
      }
      localStorage.setItem('lavera_admin_pass', newPass);
      document.getElementById('new-admin-pass').value = '';
      alert("تم تحديث كلمة المرور بنجاح.");
    }

        localStorage.removeItem('lavera_products_data');
        location.reload();
      }
    }
